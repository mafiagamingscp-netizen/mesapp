حل زرار "اختيار صور/فيديو" اللي مش بيشتغل جوه Sketchware (WebView)
=====================================================================
السبب: WebView بيتجاهل <input type="file"> إلا لو التطبيق نفسه فيه onShowFileChooser.

1) في Sketchware: Activity الخاصة بالـWebView > More Blocks / Add Source Directly
   ضيف الكود ده (webview1 = اسم الـWebView عندك):

// ---- فوق (Fields) ----
private android.webkit.ValueCallback<android.net.Uri[]> lvxFileCb;
private static final int LVX_FILE_REQ = 7788;

// ---- بعد إعداد الـWebView (onCreate / initializeLogic) ----
webview1.getSettings().setJavaScriptEnabled(true);
webview1.getSettings().setDomStorageEnabled(true);
webview1.getSettings().setAllowFileAccess(true);
webview1.getSettings().setMediaPlaybackRequiresUserGesture(false);
webview1.setWebChromeClient(new android.webkit.WebChromeClient() {
  @Override
  public boolean onShowFileChooser(android.webkit.WebView w, android.webkit.ValueCallback<android.net.Uri[]> cb, FileChooserParams params) {
    if (lvxFileCb != null) lvxFileCb.onReceiveValue(null);
    lvxFileCb = cb;
    android.content.Intent i = new android.content.Intent(android.content.Intent.ACTION_GET_CONTENT);
    i.addCategory(android.content.Intent.CATEGORY_OPENABLE);
    i.setType("*/*");
    i.putExtra(android.content.Intent.EXTRA_MIME_TYPES, new String[]{"image/*","video/*","audio/*"});
    i.putExtra(android.content.Intent.EXTRA_ALLOW_MULTIPLE, true);
    startActivityForResult(android.content.Intent.createChooser(i, "اختر ملفات"), LVX_FILE_REQ);
    return true;
  }
  @Override
  public void onPermissionRequest(final android.webkit.PermissionRequest r) {
    runOnUiThread(new Runnable(){ public void run(){ r.grant(r.getResources()); }});
  }
});

// ---- onActivityResult ----
@Override
protected void onActivityResult(int req, int res, android.content.Intent data) {
  super.onActivityResult(req, res, data);
  if (req == LVX_FILE_REQ) {
    android.net.Uri[] results = null;
    if (res == RESULT_OK && data != null) {
      if (data.getClipData() != null) {
        int n = data.getClipData().getItemCount();
        results = new android.net.Uri[n];
        for (int k = 0; k < n; k++) results[k] = data.getClipData().getItemAt(k).getUri();
      } else if (data.getData() != null) {
        results = new android.net.Uri[]{ data.getData() };
      }
    }
    if (lvxFileCb != null) { lvxFileCb.onReceiveValue(results); lvxFileCb = null; }
  }
}

2) Permissions في Sketchware (Manifest): READ_EXTERNAL_STORAGE, CAMERA, RECORD_AUDIO
   (وعلى أندرويد 13+ اطلب READ_MEDIA_IMAGES و READ_MEDIA_VIDEO).
3) لو بتجرب على المتصفح العادي (Chrome) من غير Sketchware، الزرار بيشتغل طبيعي.
